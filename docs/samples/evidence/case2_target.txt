import math


def average(values):
    total = 0.0
    for v in values:
        total += v
    return total / len(values)


def sample_variance(values):
    m = average(values)
    total = 0.0
    for v in values:
        diff = v - m
        total += diff * diff
    return total / (len(values) - 1)


def sample_stddev(values):
    return math.sqrt(sample_variance(values))
